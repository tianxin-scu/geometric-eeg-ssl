\begin{frame}{Problem: EEG is non-stationary -- and the geometry is what changes}
  \vspace{-0.4em}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \small
      EEG drifts on \textbf{three geometric axes}:
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item \textbf{Session} — cap shifts day to day.
        \item \textbf{Subject} — head sizes differ; \texttt{C3} is not the same point.
        \item \textbf{Montage} — 2 to 256 electrodes in different layouts.
      \end{itemize}
      \vspace{0.4em}
      \footnotesize
      \textcolor{codexcolor}{\textbf{Codex models}} (LaBraM, EEGPT):
      identity-indexed embeddings --- \textbf{cannot even run} on an unseen layout.
      \textcolor{chindcolor}{\textbf{Channel-independent}} (BIOT):
      no spatial structure at all.
      \\[0.3em]
      Neither uses scalp geometry as an \emph{input}.
    \end{column}
    \begin{column}{0.56\textwidth}
      \begin{tikzpicture}[scale=0.85, every node/.style={font=\scriptsize}]
        % Three heads side by side showing the three drift axes
        \foreach \x/\lbl/\sub in {0/Session/{day 1 vs day 2}, 3.6/Subject/{small vs large head}, 7.2/Montage/{64 ch vs 3 ch}} {
          \draw[thick] (\x,0) circle (1.2);
          \node at (\x,-1.6) {\textbf{\lbl}};
          \node at (\x,-1.95) {\sub};
        }
        % Session: shifted electrodes
        \foreach \a in {30, 90, 150, 210, 270, 330} {
          \fill[geocolor]   ({0  + 1.0*cos(\a)}, {0 + 1.0*sin(\a)}) circle (0.07);
          \fill[accentred!70] ({0  + 1.0*cos(\a+12)}, {0 + 1.0*sin(\a+12)}) circle (0.07);
        }
        % Subject: small head outline vs large; same labels, different scale
        \draw[thick, dashed] (3.6,0) circle (0.85);
        \foreach \a in {0, 60, 120, 180, 240, 300} {
          \fill[geocolor] ({3.6 + 1.0*cos(\a)}, {0 + 1.0*sin(\a)}) circle (0.07);
          \fill[codexcolor] ({3.6 + 0.7*cos(\a)}, {0 + 0.7*sin(\a)}) circle (0.07);

% Slide 3 — Solution: geometry-conditioned attention
% =============================================================================
\begin{frame}{Solution: spatial attention conditioned on $g_{ij}$}
  \vspace{-0.3em}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \small
      \textbf{Geometric descriptor.}
      \[
        g_{ij} = \big[\,p_i,\; p_j,\; p_i - p_j,\; \|p_i - p_j\|\,\big] \in \mathbb{R}^{10}
      \]
      Positions + displacement + distance. No per-electrode parameters; same shared MLP for \emph{any} montage.
      \\[0.4em]
      \textbf{Injection fixed to G3} (geometry at \emph{both} score and value -- strongest in my earlier ablation). I instead ablate the descriptor: \textbf{pos\_only} $=[p_i,p_j]\in\mathbb{R}^6$.
      \\[0.4em]
      \footnotesize
      The only rival that can \emph{also} transfer is \textcolor{chindcolor}{\textbf{chind}} (no geometry); the codex \textbf{cannot be evaluated} zero-shot.
    \end{column}
    \begin{column}{0.56\textwidth}
      \begin{tikzpicture}[scale=0.78, every node/.style={font=\scriptsize}, transform shape]
        % Left -- transductive
        \node[draw, fill=codexcolor!15, rounded corners, minimum width=3.2cm, minimum height=0.7cm] (cT) at (0,3.5) {\textbf{Transductive (codex)}};
        \node[draw, fill=codexcolor!8, rounded corners, minimum width=3.2cm, minimum height=0.6cm] (cI) at (0,2.6) {\texttt{embed[C3]} $+$ patch};
        \node[draw, fill=codexcolor!8, rounded corners, minimum width=3.2cm, minimum height=0.6cm] (cQK) at (0,1.7) {dot-product attn $QK^\top$};
        \node[draw, fill=accentred!15, rounded corners, minimum width=3.2cm, minimum height=0.6cm, align=center] (cFail) at (0,0.6) {\scriptsize \emph{cannot transfer}\\ \scriptsize no slot for unseen \texttt{C3}};
        \draw[->, thick] (cI) -- (cQK);
        \draw[->, thick] (cQK) -- (cFail);

        % Right -- geometric
        \node[draw, fill=geocolor!15, rounded corners, minimum width=3.2cm, minimum height=0.7cm] (gT) at (5,3.5) {\textbf{Geometric (G3)}};
        \node[draw, fill=geocolor!8, rounded corners, minimum width=3.2cm, minimum height=0.6cm] (gI) at (5,2.6) {patch $\;\&\;$ $g_{ij}$};
        \node[draw, fill=geocolor!8, rounded corners, minimum width=3.2cm, minimum height=0.6cm] (gA) at (5,1.7) {attn MLP$(x_i, x_j, g_{ij})$};
        \node[draw, fill=winnercolor!18, rounded corners, minimum width=3.2cm, minimum height=0.6cm, align=center] (gOK) at (5,0.6) {\scriptsize \emph{transfers to any montage}\\ \scriptsize given only $p_i$};
        \draw[->, thick] (gI) -- (gA);
        \draw[->, thick] (gA) -- (gOK);

        % Training pipeline ribbon
        \node[font=\scriptsize] at (2.5,-0.5) {raw EEG $\to$ patch $\to$ spatial attn $\to$ temporal $\to$ momentum target (EMA)};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Experiment: leave-one-montage-out, zero-shot transfer}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \small
      Train on a \textbf{mix of montages}; hold out one entire dataset; freeze the encoder; fit a linear probe on the \textbf{held-out montage} (genuinely zero-shot).
      \\[0.5em]
      Identical training for every variant --- only the spatial encoder differs (fair comparison).
      \\[0.5em]
      \footnotesize
      Balanced 9-subject-per-dataset corpus: geometric $\geq$ \textbf{chind} on \textbf{3/3} held-out montages. Direction is consistent; gaps are within noise at this scale.
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \footnotesize \textbf{Zero-shot BAC (balanced 9-subj corpus)}\\[0.3em]
      \begin{tabular}{lccc}
        \toprule
        Held-out & \textcolor{geocolor}{\textbf{geom}} & \textcolor{posonlycolor}{\textbf{pos\_only}} & \textcolor{chindcolor}{\textbf{chind}} \\
        \midrule
        Sleep-EDFx    & 0.593 & \textbf{0.597} & 0.589 \\
        BCIC-2B       & \textbf{0.509} & 0.508 & 0.504 \\
        PhysioNet MI  & 0.303 & \textbf{0.315} & 0.302 \\
        \bottomrule
      \end{tabular}
      \\[0.6em]
      \scriptsize
      Different tasks $\Rightarrow$ different chance levels; read \emph{within} each row (geom vs chind), not across rows. Consistent but not yet significant --- motivates scaling the one split we can.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Headline: at scale, geometric beats chind on zero-shot transfer}
  \vspace{-0.2em}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \centering
      \begin{tikzpicture}
        \begin{axis}[
          width=7.6cm, height=5.0cm,

          xmin=0.49, xmax=0.57,
          symbolic y coords={chind, posonly, full},
          ytick=data,
          yticklabels={chind, geom (pos\_only), geom (full)},

          nodes near coords style={font=\tiny},

          \addplot[fill=chindcolor!60]   coordinates {(0.509,chind)};
          \addplot[fill=posonlycolor!80] coordinates {(0.531,posonly)};
          \addplot[fill=geocolor!80]     coordinates {(0.545,full)};
          % Chance line at 0.50
          \draw[dashed, gray] (axis cs:0.50,chind) -- (axis cs:0.50,full);
        \end{axis}
      \end{tikzpicture}\\[-0.1em]
      \scriptsize Chance = 0.50 (binary). Dashed line = chance.
    \end{column}
    \begin{column}{0.42\textwidth}
      \small
      \textbf{Scaled the one feasible split:}\\
      \emph{balanced n=156 (PhysioNet 78 + Sleep 78), zero-shot to BCIC-2B.}
      \\[0.5em]
      \textcolor{geocolor}{\textbf{Geometric (full) = 0.545}} beats chind (0.509):
      \textbf{+0.036}, $p=0.023$, 7/9 subjects.
      \\[0.4em]
      \textcolor{posonlycolor}{\textbf{pos\_only}} \emph{also} beats chind:
      \textbf{+0.022}, $p=0.013$, 8/9 subjects.
      \\[0.4em]
      \footnotesize Codex is absent --- it \emph{cannot} be evaluated here at all. \textit{That absence is the point.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusion: a clean positive, with clear edges}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small
      \textbf{What I found:}
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item Geometry-conditioned attention \textbf{transfers zero-shot} to an unseen montage and \textbf{beats chind significantly} once given enough data.
        \item Robust to descriptor reduction (pos\_only also significant).
      \end{itemize}
      \vspace{0.3em}
      \textbf{Honest limits:}
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item Significant on \textbf{one} held-out montage (BCIC); small-corpus tests directional but underpowered (9-subject ceiling).
      \end{itemize}
      \vspace{0.3em}
      \textbf{Future work:} second distinct-montage held-out (e.g.\ 128\,ch); larger corpus; implement the codex baseline.
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
