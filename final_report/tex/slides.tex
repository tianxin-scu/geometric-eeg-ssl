\begin{frame}{Problem: EEG is non-stationary -- and the geometry is what changes}
  \vspace{-0.4em}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \small
      EEG varies on \textbf{three geometric axes}:
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item \textbf{Session} — cap shifts day to day. \emph{(motivates)}
        \item \textbf{Subject} — head sizes differ. \emph{(motivates)}
        \item \textbf{Montage} — 2 to 256 electrodes in different layouts. \textbf{$\leftarrow$ what I test}
      \end{itemize}
      \vspace{0.4em}
      \footnotesize
      \textcolor{codexcolor}{\textbf{Codex models}} (LaBraM, EEGPT):
      identity-indexed embeddings --- \textbf{cannot even run} on an unseen layout.
      \textcolor{chindcolor}{\textbf{Channel-independent}} (BIOT):
      no spatial structure at all.
      \\[0.3em]
      Neither uses scalp geometry as an \emph{input}.
    \end{column}

\begin{frame}{Solution: spatial attention conditioned on $g_{ij}$}
  \vspace{-0.3em}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \small
      \[
        g_{ij} = \bigl[\,p_i,\;\; p_j,\;\; p_i - p_j,\;\; \|p_i - p_j\|\,\bigr] \in \mathbb{R}^{10}
      \]
      \vspace{0.1em}
      \footnotesize
      \textbf{Example:} $g_{\texttt{C3}\to\texttt{Cz}}$ (normalized coords)
      \\[0.3em]
      \renewcommand{\arraystretch}{1.25}
      \begin{tabular}{@{}llr@{}}
        \toprule
        $p_i$ (\texttt{C3})    & $(x,y,z)$ & $(-0.71,\,0.00,\,0.59)$ \\
        $p_j$ (\texttt{Cz})    & $(x,y,z)$ & $(\phantom{-}0.00,\,0.00,\,0.95)$ \\
        $p_i - p_j$            & disp.     & $(-0.71,\,0.00,\,-0.36)$ \\
        $\|p_i - p_j\|$        & dist.     & $0.80$ \\
        \bottomrule
      \end{tabular}
      \\[0.4em]
      \scriptsize $3+3+3+1 = 10$ dims. \textbf{pos\_only} ablation keeps just $[p_i,p_j]\in\mathbb{R}^6$.
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \begin{tikzpicture}[
          >=Stealth,
          every node/.style={font=\scriptsize},
          box/.style={draw, rounded corners, minimum height=0.5cm, minimum width=1.1cm, align=center, inner sep=3pt},
          op/.style={draw, circle, fill=geocolor!18, minimum size=0.5cm, inner sep=0pt},
          mlp/.style={draw, rounded corners, fill=geocolor!25, minimum height=0.5cm, inner sep=3pt},
          scale=1.0, transform shape,
        ]
        % --- main attention chain (x = 0) ---
        \node[box, fill=gray!10]        (patch) at (0, 4.4) {patch};
        \node[box, fill=geocolor!12]    (qkv)   at (0, 3.5) {$q,\,k,\,v$};
        \node[box]                      (att)   at (0, 2.6) {dot-prod};
        \node[op]                       (plus)  at (0, 1.8) {$+$};
        \node[box]                      (soft)  at (0, 1.0) {softmax};
        \node[op]                       (dot)   at (0, 0.0) {$\cdot$};
        \node[box, fill=gray!10]        (out)   at (0,-0.9) {output};

        % --- geometry side (right) ---
        \node[box, fill=geocolor!12]    (gij)   at (2.7, 3.95) {$g_{ij}$};
        \node[mlp, fill=biaspathcol!35]  (bmlp)  at (2.7, 2.6)  {bias MLP};
        \node[mlp, fill=valuepathcol!30] (wmlp)  at (2.7, 1.0)  {value MLP};
        \node[op]                       (vsum)  at (1.35, 0.0) {$+$};

        % --- main chain arrows ---
        \draw[->] (patch) -- (qkv);
        \draw[->] (qkv) -- (att) node[midway, right, font=\tiny] {$q,k$};
        \draw[->] (att) -- (plus);
        \draw[->] (plus) -- (soft);
        \draw[->] (soft) -- (dot) node[midway, left, font=\tiny] {$a_{ij}$};
        \draw[->] (vsum) -- (dot);
        \draw[->] (dot) -- (out);

        % --- geometry: g_ij feeds BOTH MLPs (parallel); bias path vs value path ---
        \draw[->, biaspathcol, thick]  (gij) -- (bmlp);
        \draw[->, valuepathcol, thick] (gij.east) to[out=-90, in=90] (wmlp.east);
        \draw[->, biaspathcol, thick]  (bmlp) -- (plus) node[midway, above, font=\tiny] {$b(g_{ij})$};
        \draw[->, valuepathcol, thick] (wmlp) -- (vsum) node[midway, above, font=\tiny] {$w(g_{ij})$};
        % v from the q,k,v box into the value sum
        \draw[->] (qkv.east) to[out=0, in=90] (1.35, 0.25) -- (vsum)
          node[pos=0.18, above, font=\tiny] {$v$};
      \end{tikzpicture}

      \vspace{0.2em}
      \scriptsize
      \textcolor{biaspathcol}{\textbf{bias path}} only $=$ G1,\quad
      \textcolor{valuepathcol}{\textbf{value path}} only $=$ G2,\quad both $=$ G3.\\
      Ablation: the signal is in the \textcolor{valuepathcol}{\textbf{value path}}
      --- G2\,$\approx$\,G3\,$\gg$\,G1 (G1 $\approx$ no geometry).
    \end{column}
  \end{columns}
\end{frame}

% Slide 5 — Conclusion + future work

\begin{frame}{Conclusion: a clean positive, with clear edges}
  \begin{columns}[T]
    \begin{column}{0.9\textwidth}
      \small
      \textbf{What I found:}
      \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item Geometry-conditioned attention \textbf{transfers zero-shot} to an unseen montage and \textbf{beats chind significantly} once given enough data.
        \item Robust to descriptor reduction (pos\_only also significant).
      \end{itemize}
      \vspace{0.5em}
      \textbf{Honest limits:}
      \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item Significant on \textbf{one} held-out montage (BCIC); small-corpus tests point the right way but are not significant (9-subject ceiling, underpowered).
      \end{itemize}
      \vspace{0.5em}
      \textbf{Future work:} second distinct-montage held-out (e.g.\ 128\,ch); larger corpus; implement the codex baseline.
    \end{column}
  \end{columns}
\end{frame}
